% --- Proof: Box-Constraint QP Reduces to Clipping ---
%
% \begin{proposition}[Box-Constraint QP Equivalence]
% \label{prop:qp-clip}
% For hyperrectangular (box) constraints $\mathcal{C} = \{\mathbf{a} \in \mathbb{R}^d : \mathbf{l} \leq \mathbf{a} \leq \mathbf{u}\}$,
% the CBF-QP that AEGIS solves:
% \begin{equation}
%   \mathbf{a}_{\mathrm{safe}} = \arg\min_{\mathbf{a}' \in \mathcal{C}} \|\mathbf{a}' - \mathbf{a}\|_2^2
%   \quad \text{s.t.} \quad l_i \leq a'_i \leq u_i, \;\; \forall\, i \in \{1, \ldots, d\}
% \end{equation}
% has the closed-form solution $\mathbf{a}_{\mathrm{safe}} = \mathrm{clip}(\mathbf{a}, \mathbf{l}, \mathbf{u})$.
% \end{proposition}
%
% \begin{proof}
% The objective $\|\mathbf{a}' - \mathbf{a}\|_2^2 = \sum_{i=1}^{d}(a'_i - a_i)^2$ is separable
% across dimensions, and the box constraints are likewise separable. Therefore the
% $d$-dimensional QP decomposes into $d$ independent scalar problems:
% \begin{equation}
%   a_{\mathrm{safe},i} = \arg\min_{l_i \leq a'_i \leq u_i} (a'_i - a_i)^2, \quad i = 1, \ldots, d.
% \end{equation}
% Each scalar problem minimizes a strictly convex quadratic on a closed interval $[l_i, u_i]$.
% The unconstrained minimizer is $a'_i = a_i$. By projection onto the interval:
% \begin{equation}
%   a_{\mathrm{safe},i} = \max\bigl(l_i,\; \min(a_i,\; u_i)\bigr) = \mathrm{clip}(a_i,\; l_i,\; u_i).
% \end{equation}
% Therefore $\mathbf{a}_{\mathrm{safe}} = \mathrm{clip}(\mathbf{a},\; \mathbf{l},\; \mathbf{u})$.
% \end{proof}
%
% \textbf{Implication.} For box constraints, SafeContract's clipping produces outputs
% \emph{mathematically identical} to AEGIS's CBF-QP solution. The distinction is in
% scope and overhead: AEGIS handles arbitrary CBF constraints (obstacle avoidance,
% scene-aware safety) at the cost of a VLM + depth sensor + QP solver (${\sim}360\,\mu$s),
% while SafeContract handles action-space constraints requiring no external perception
% (${\sim}13\,\mu$s). The approaches are complementary.
